local Connections = {}
local function addConn(name, conn)
	if Connections[name] then
		pcall(function() Connections[name]:Disconnect() end)
	end
	Connections[name] = conn
end
local function stopConn(name)
	if Connections[name] then
		pcall(function() Connections[name]:Disconnect() end)
		Connections[name] = nil
	end
end

local function ForceStopAll()
	for k, v in pairs(State) do
		if type(v) == "boolean" then State[k] = false end
	end
	for name, _ in pairs(Connections) do
		stopConn(name)
	end
	-- 追加クリーンアップ
	pcall(function()
		local char = LP.Character
		if char then
			for _, v in ipairs(char:GetDescendants()) do
				if v:IsA("BodyVelocity") or v:IsA("BodyPosition") or v:IsA("BodyAngularVelocity") or v:IsA("BodyForce") then
					if not v.Name:match("Allowed") then v:Destroy() end
				end
			end
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum then
				hum.WalkSpeed = 16
				hum.JumpPower = 50
			end
		end
	end)
	Notify("強制停止", "全機能を停止しました", 3)
end

local GrabEvents = RS:FindFirstChild("GrabEvents")
local CreateGrabLine = GrabEvents and GrabEvents:FindFirstChild("CreateGrabLine")
local DestroyGrabLine = GrabEvents and GrabEvents:FindFirstChild("DestroyGrabLine")
local SetNetworkOwner = GrabEvents and GrabEvents:FindFirstChild("SetNetworkOwner")

local function spamOwnership(hrp)
	if SetNetworkOwner and hrp then
		pcall(function() SetNetworkOwner:FireServer(hrp, hrp.CFrame) end)
	end
end

local function destroyLineOnPlayer(hrp)
	if CreateGrabLine and DestroyGrabLine and hrp then
		pcall(function()
			CreateGrabLine:FireServer(hrp, CFrame.new(0, 1e9, 0))
			task.wait()
			DestroyGrabLine:FireServer(hrp)
		end)
	end
end

local function GetPlayerHRP(player)
	if not player then return nil end
	local char = player.Character
	if char and char.Parent == WS then
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if hrp then return hrp end
	end
	local plotItems = WS:FindFirstChild("PlotItems")
	if plotItems then
		for _, plot in ipairs(plotItems:GetChildren()) do
			if plot:IsA("Folder") then
				local playersInPlot = plot:FindFirstChild("PlayersInPlots")
				if playersInPlot and playersInPlot:FindFirstChild(player.Name) then
					for _, child in ipairs(plot:GetDescendants()) do
						if child:IsA("Model") and child.Name == player.Name then
							local hrp = child:FindFirstChild("HumanoidRootPart")
							if hrp then return hrp end
						end
					end
				end
			end
		end
	end
	for _, model in ipairs(WS:GetDescendants()) do
		if model:IsA("Model") and model.Name == player.Name then
			local hrp = model:FindFirstChild("HumanoidRootPart")
			if hrp then return hrp end
		end
	end
	return nil
end

local function getPlayerNameList()
	local list = {}
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP then table.insert(list, plr.Name) end
	end
	return list
end

local function findPlayerByName(name)
	if not name or name == "" then return nil end
	local lower = name:lower()
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Name:lower() == lower or plr.DisplayName:lower() == lower then return plr end
	end
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Name:lower():find(lower, 1, true) or plr.DisplayName:lower():find(lower, 1, true) then return plr end
	end
	return nil
end

-- 1. ターゲットラグ実行
-- ==============================
local function executeTargetLag()
	Notify("ターゲットラグ", "実行開始", 3)
	task.spawn(function()
		local players = {}
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= LP then table.insert(players, plr) end
		end
		if #players == 0 then
			Notify("ターゲットラグ", "対象なし", 2)
			return
		end
		local myChar = LP.Character
		local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
		if not myHrp then return end

		-- ラグスパム開始
		local lagConn
		lagConn = RUN.Heartbeat:Connect(function()
			local spawn = WS:FindFirstChild("SpawnLocation") or WS:FindFirstChild("Spawn") or myHrp
			if spawn and CreateGrabLine then
				for i = 1, 4 do
					pcall(function()
						CreateGrabLine:FireServer(spawn, CFrame.new(math.random(-2e9, 2e9), 0, math.random(-2e9, 2e9)))
					end)
				end
			end
		end)

		local height = 15
		local center = Vector3.new(0, height, 0)
		local spawnLoc = WS:FindFirstChild("SpawnLocation") or WS:FindFirstChild("Spawn")
		if spawnLoc then center = Vector3.new(spawnLoc.Position.X, height, spawnLoc.Position.Z) end

		myHrp.CFrame = CFrame.new(center)
		myHrp.AssemblyLinearVelocity = Vector3.zero

		for _, plr in ipairs(players) do
			local hrp = GetPlayerHRP(plr)
			if hrp then
				myHrp.CFrame = hrp.CFrame * CFrame.new(0, 5, 5)
				spamOwnership(hrp)
				task.wait(0.1)
			end
		end

		local radius = 15
		local angleStep = (math.pi * 2) / math.max(#players, 1)
		for i, plr in ipairs(players) do
			local hrp = GetPlayerHRP(plr)
			if hrp then
				local angle = (i - 1) * angleStep
				local x = math.cos(angle) * radius
				local z = math.sin(angle) * radius
				pcall(function()
					hrp.CFrame = CFrame.new(center.X + x, height, center.Z + z)
					hrp.AssemblyLinearVelocity = Vector3.zero
					local bp = Instance.new("BodyPosition")
					bp.MaxForce = Vector3.new(9e9, 9e9, 9e9)
					bp.P = 5e10
					bp.Position = Vector3.new(center.X + x, height, center.Z + z)
					bp.Parent = hrp
					task.delay(2, function() pcall(function() bp:Destroy() end) end)
				end)
			end
		end

		for i = 1, 8 do
			for _, plr in ipairs(players) do
				local hrp = GetPlayerHRP(plr)
				if hrp then destroyLineOnPlayer(hrp) end
			end
			task.wait(0.3)
		end

		task.wait(5)
		if lagConn then lagConn:Disconnect() end
		Notify("ターゲットラグ", "完了", 2)
	end)
end

-- 2. 超高速ラグパネル (1200本)
-- ==============================
local lineLagEnabled = false
local lineLagThread = nil
local function startLineLag()
	if lineLagEnabled then return end
	lineLagEnabled = true
	lineLagThread = coroutine.create(function()
		while lineLagEnabled do
			local spawn = WS:FindFirstChild("SpawnLocation") or WS:FindFirstChild("Spawn") or (LP.Character and LP.Character:FindFirstChild("HumanoidRootPart"))
			if spawn and CreateGrabLine then
				for i = 1, 1200 do
					pcall(function()
						CreateGrabLine:FireServer(spawn, CFrame.new(math.random(-1e9, 1e9), 0, math.random(-1e9, 1e9)))
					end)
				end
			end
			task.wait(0.016)
		end
	end)
	coroutine.resume(lineLagThread)
	Notify("超高速ラグパネル", "1200本開始", 2)
end
local function stopLineLag()
	lineLagEnabled = false
	if lineLagThread then pcall(function() coroutine.close(lineLagThread) end) lineLagThread = nil end
	Notify("超高速ラグパネル", "停止", 2)
end

local function restartMap()
	pcall(function()
		local eventsToTry = {"ResetMap", "RestartMap", "MapReset", "ServerReset", "RecreateMap"}
		local triggered = false
		for _, eventName in ipairs(eventsToTry) do
			local remote = RS:FindFirstChild(eventName, true)
			if remote and (remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent")) then
				remote:FireServer()
				triggered = true
			elseif remote and remote:IsA("RemoteFunction") then
				task.spawn(function() pcall(function() remote:InvokeServer() end) end)
				triggered = true
			end
		end
		local char = LP.Character
		if char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if hum and hrp then
				local spawn = WS:FindFirstChild("SpawnLocation") or WS:FindFirstChild("Spawn")
				if spawn then
					hrp.CFrame = spawn.CFrame + Vector3.new(0, 5, 0)
					hrp.AssemblyLinearVelocity = Vector3.zero
				end
			end
		end
		Notify("マップ再起動", triggered and "シグナル送信" or "位置リセット", 3)
	end)
end

-- 6. Fly（トグル + 速度スライダー）
-- ==============================
local flyBg, flyBv
local function startFly()
	State.FlyEnabled = true
	local char = LP.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	flyBg = Instance.new("BodyGyro")
	flyBg.P = 9e4
	flyBg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	flyBg.CFrame = hrp.CFrame
	flyBg.Parent = hrp
	flyBv = Instance.new("BodyVelocity")
	flyBv.Velocity = Vector3.zero
	flyBv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	flyBv.Parent = hrp
	addConn("Fly", RUN.RenderStepped:Connect(function()
		if not State.FlyEnabled or not hrp or not hrp.Parent then return end
		local cam = WS.CurrentCamera
		local dir = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
		if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.C) then dir = dir - Vector3.new(0, 1, 0) end
		-- モバイル対応
		if UIS.TouchEnabled then
			-- 簡易タッチ対応（カメラ方向に前進）
		end
		if dir.Magnitude > 0 then dir = dir.Unit * State.FlySpeed end
		flyBv.Velocity = dir
		flyBg.CFrame = cam.CFrame
	end))
	Notify("Fly", "有効 速度" .. State.FlySpeed, 2)
end
local function stopFly()
	State.FlyEnabled = false
	stopConn("Fly")
	if flyBg then flyBg:Destroy() flyBg = nil end
	if flyBv then flyBv:Destroy() flyBv = nil end
	Notify("Fly", "無効", 2)
end

local ESPFolder = nil
local function createESP(plr)
	pcall(function()
		if not plr or plr == LP or not plr.Character then return end
		if plr.Character:FindFirstChild("NEKO_ESP") then return end
		local hl = Instance.new("Highlight")
		hl.Name = "NEKO_ESP"
		hl.FillColor = Color3.fromRGB(255, 0, 0)
		hl.OutlineColor = Color3.fromRGB(255, 255, 255)
		hl.FillTransparency = 0.5
		hl.OutlineTransparency = 0
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.Parent = plr.Character

		local bill = Instance.new("BillboardGui")
		bill.Name = "NEKO_ESP_Bill"
		bill.Size = UDim2.new(0, 100, 0, 40)
		bill.StudsOffset = Vector3.new(0, 3, 0)
		bill.AlwaysOnTop = true
		bill.Parent = plr.Character:FindFirstChild("Head") or plr.Character:FindFirstChild("HumanoidRootPart")
		local label = Instance.new("TextLabel", bill)
		label.Size = UDim2.new(1, 0, 1, 0)
		label.BackgroundTransparency = 1
		label.TextColor3 = Color3.new(1, 1, 1)
		label.TextStrokeTransparency = 0.5
		label.Font = Enum.Font.GothamBold
		label.TextSize = 14
		label.Text = plr.Name
		task.spawn(function()
			while bill.Parent and State.ESPEnabled do
				local myHrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
				local th = GetPlayerHRP(plr)
				if myHrp and th then
					local dist = math.floor((th.Position - myHrp.Position).Magnitude)
					label.Text = plr.Name .. "\n" .. dist .. " studs"
				end
				task.wait(0.5)
			end
		end)
	end)
end
local function removeESP(plr)
	pcall(function()
		if plr and plr.Character then
			local hl = plr.Character:FindFirstChild("NEKO_ESP")
			if hl then hl:Destroy() end
			local head = plr.Character:FindFirstChild("Head") or plr.Character:FindFirstChild("HumanoidRootPart")
			if head then
				local bill = head:FindFirstChild("NEKO_ESP_Bill")
				if bill then bill:Destroy() end
			end
		end
	end)
end
local function startESP()
	State.ESPEnabled = true
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP then createESP(plr) end
	end
	addConn("ESP_Add", Players.PlayerAdded:Connect(function(plr)
		task.wait(1)
		if State.ESPEnabled then createESP(plr) end
	end))
	addConn("ESP_Remove", Players.PlayerRemoving:Connect(function(plr)
		removeESP(plr)
	end))
	Notify("ESP", "有効", 2)
end
local function stopESP()
	State.ESPEnabled = false
	stopConn("ESP_Add")
	stopConn("ESP_Remove")
	for _, plr in ipairs(Players:GetPlayers()) do
		removeESP(plr)
	end
	Notify("ESP", "無効", 2)
end

-- 35. ゲーム全UI色変更（オレンジ→緑グラデ）
-- ==============================
local GameUIOriginals = {}
local function startGameUIColor()
	State.GameUIColorEnabled = true
	local ORANGE = Color3.fromRGB(255, 140, 0)
	local GREEN = Color3.fromRGB(20, 200, 92)
	local function IsOrange(c)
		if not c then return false end
		local r, g, b = c.R, c.G, c.B
		return r > 0.55 and g >= 0.15 and g < 0.85 and b < 0.5
	end
	local function apply(obj)
		if obj:IsA("GuiObject") then
			if IsOrange(obj.BackgroundColor3) then
				if not GameUIOriginals[obj] then GameUIOriginals[obj] = obj.BackgroundColor3 end
				obj.BackgroundColor3 = GREEN
			end
			if obj:IsA("TextLabel") or obj:IsA("TextButton") then
				if IsOrange(obj.TextColor3) then
					if not GameUIOriginals[obj] then GameUIOriginals[obj] = obj.TextColor3 end
					obj.TextColor3 = GREEN
				end
			end
		end
	end
	for _, gui in ipairs(CG:GetDescendants()) do apply(gui) end
	for _, gui in ipairs(LP.PlayerGui:GetDescendants()) do apply(gui) end
	addConn("GameUI", LP.PlayerGui.DescendantAdded:Connect(function(obj) if State.GameUIColorEnabled then apply(obj) end end))
	Notify("ゲームUI色変更", "有効", 2)
end
local function stopGameUIColor()
	State.GameUIColorEnabled = false
	stopConn("GameUI")
	for obj, col in pairs(GameUIOriginals) do
		pcall(function()
			if obj:IsA("GuiObject") then obj.BackgroundColor3 = col end
		end)
	end
	GameUIOriginals = {}
	Notify("ゲームUI色変更", "無効", 2)
end

local function serverHop()
	Notify("サーバーホップ", "別サーバーへ移動中...", 3)
	pcall(function()
		local servers = {}
		local req = request or http_request or (syn and syn.request)
		if req then
			local res = req({
				Url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
			})
			if res and res.Body then
				local data = HttpService:JSONDecode(res.Body)
				if data and data.data then
					for _, s in ipairs(data.data) do
						if s.playing < s.maxPlayers and s.id ~= game.JobId then
							table.insert(servers, s.id)
						end
					end
				end
			end
		end
		if #servers > 0 then
			TPS:TeleportToPlaceInstance(game.PlaceId, servers[math.random(1, #servers)], LP)
		else
			TPS:Teleport(game.PlaceId, LP)
		end
	end)
end

function LoadMusicVersion()

	local Tab = W:AddTab("音楽プレイヤー")
	local Gp = Tab:AddLeftGroupbox("曲リスト")
	for _, song in ipairs(MusicList) do
		Gp:AddButton({
			Text = song.Name,
			Func = function()
				if currentSound then currentSound:Stop() currentSound:Destroy() end
				currentSound = Instance.new("Sound")
				currentSound.SoundId = "rbxassetid://" .. song.Id
				currentSound.Volume = 1
				currentSound.Looped = true
				currentSound.Parent = SS
				currentSound:Play()
				H:Notify("再生中: " .. song.Name, 2)
			end
		})
	end
	local Ctrl = Tab:AddLeftGroupbox("コントロール")
	Ctrl:AddButton({Text = "停止", Func = function()
		if currentSound then currentSound:Stop() currentSound:Destroy() currentSound = nil end

	end})

		if currentSound then currentSound:Stop() currentSound:Destroy() end
		W.Gui:Destroy()
		CreateVersionSelector()
	end})
